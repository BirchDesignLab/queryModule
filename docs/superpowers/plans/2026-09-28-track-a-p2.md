## Global Constraints

- Runtime: Node 24 LTS per ADR-0001; TypeScript strict everywhere; no `any`.
- Workspace: pnpm workspace; `pnpm install --frozen-lockfile` everywhere. No new runtime dependency in this plan; a dependency addition would go in its own `chore/deps-<name>` PR.
- Method: TDD for every task; the failing test is written and seen failing before code (master plan 6.3 step 12). Tell the implementer up front that TDD order is reviewed.
- Git: one wave, one branch `feat/a-p2-wave-<k>`, one PR, squash merge; contract tasks each get their own `feat/core-<n>` PR (master plan 8.1). Never commit to `main`. Ask the developer before every push.
- Contract first: every change under `packages/core/src/contracts/**`, `packages/core/src/config/**` schema files or `packages/api/openapi.json` is a developer decision (listed at the end) and lands in its own `contract` PR before any consumer (master plan 8).
- Core is pure: no IO, no clock reads, no randomness in `packages/core` (`now` is passed in).
- Persistence: timestamps are INTEGER epoch ms UTC; ids UUIDv7 text (spec 5.5); durations from a monotonic clock (spec 4.7).
- Audit: every audit write goes through `AuditService.record(tx, event)` inside the transaction that makes the change; any audit failure rolls the transaction back and the request answers 500 `internal` (spec 5.2, fail closed). Audit `details` hold identifiers, metadata and `role: "type"` values only.
- Values: persisted and dispatched values are built only from the server `FormState`'s visible effective values; the raw body is never persisted or logged (spec 5.2 step 2, 5.9).
- Errors: every API error uses `ApiError`; params never carry submitted values (spec 4.7).
- Logs: no query values, secrets, key material, DEKs, passwords or session tokens in any log line (spec 5.9).
- Fixture policy (spec 5.4): plates `ZZ-####`, synthetic names (Testerson, Sampleworth), DOBs in 1901, VINs failing the ISO 3779 check digit, addresses on Example Ave. Trigger inputs such as `ABC123` are exempt.
- Docs: no em dashes; dates MM-DD-YY in prose, ISO in code and file names.
- Windows lane: API tests use `openTempDatabase`; win32 teardown is best-effort (AV holds libsql handles); Git Bash is first on PATH.
- CI and deploy facts to keep: runs on `main` never cancel each other and every `main` push builds and publishes an image (#230, #234; a spec 9.3 deviation the developer ruled), so no task here reintroduces `cancel-in-progress` for `main` in `ci.yml`; `deploy/compose.yml` keeps `autoheal: "False"` with the capital F (#248); `restore-test.sh` keeps per-file secret mounts (#249).
- Host access: none from a session. A step that needs the deploy host is a developer-run step with exact commands; promoting is `gh workflow run promote.yml -f sha=<full sha>` from Windows, then the deploy-pull timer applies it within five minutes.

```ts
// packages/api/src/keys/aead.ts
import { createCipheriv, createDecipheriv, randomBytes } from "node:crypto";

export interface Sealed { ciphertext: Buffer; iv: Buffer; tag: Buffer }
export class AeadError extends Error {
  constructor() { super("authenticated decryption failed"); this.name = "AeadError"; }
}
const KEY_BYTES = 32;

export function seal(key: Buffer, plaintext: Buffer, aad: string): Sealed {
  if (key.length !== KEY_BYTES) throw new AeadError();
  const iv = randomBytes(12);
  const c = createCipheriv("aes-256-gcm", key, iv).setAAD(Buffer.from(aad, "utf8"));
  const ciphertext = Buffer.concat([c.update(plaintext), c.final()]);
  return { ciphertext, iv, tag: c.getAuthTag() };
}

export function open(key: Buffer, s: Sealed, aad: string): Buffer {
  try {
    if (key.length !== KEY_BYTES) throw new Error();
    const d = createDecipheriv("aes-256-gcm", key, s.iv).setAAD(Buffer.from(aad, "utf8"));
    d.setAuthTag(s.tag);
    return Buffer.concat([d.update(s.ciphertext), d.final()]);
  } catch {
    throw new AeadError();
  }
}
```

```ts
// packages/core/src/planner/plan.ts
import type { AuditValidationError } from "../contracts/audit";
import { MAX_SOURCES_PER_SUBMIT } from "../config/validate-rules";
import type { ValidationError } from "../contracts/validation-error";
import { findQueryType } from "../rules/compile";
import { evaluateForm } from "../rules/evaluate-form";
import type { CanonicalValue, EvaluateOptions, FormInput, FormMode, FormState, RulesConfig } from "../rules/types";

export function isPlanError(r: Plan | PlanError): r is PlanError {
  return "errors" in r;
}

function typeValuesOf(s: FormState): Record<string, string> {
  const out: Record<string, string> = {};
  for (const f of s.fields) if (f.role === "type" && f.visible && typeof f.effectiveValue === "string") out[f.key] = f.effectiveValue;
  return out;
}

function auditReason(e: ValidationError): AuditValidationError {
  const p = e.params ?? {};
  const params: AuditValidationError["params"] = {};
  if (typeof p.field === "string") params.field = p.field;
  if (typeof p.labelKey === "string") params.labelKey = p.labelKey;
  if (typeof p.position === "number") params.position = p.position;
  return Object.keys(params).length > 0 ? { key: e.key, params } : { key: e.key };
}

/** Spec 4.6 step 3: narrow to eligible plateOnly sources; empty is an error for the caller to map. */
function narrow(s: FormState, selected: readonly string[]): { keep: string[]; dropped: string[] } {
  if (s.mode !== "plateOnly") return { keep: [...selected], dropped: [] };
  const plateOnly = new Set(s.sources.filter((x) => x.plateOnly).map((x) => x.sourceId));
  return { keep: selected.filter((id) => plateOnly.has(id)), dropped: selected.filter((id) => !plateOnly.has(id)) };
}

export function planRequest(
  config: RulesConfig,
  queryType: string,
  userValues: FormInput,
  selectedSourceIds: readonly string[],
  options: EvaluateOptions,
): Plan | PlanError {
  // Step 1: primary.
  const primary = evaluateForm(config, queryType, userValues, options);
  if (primary.errors.length > 0) return { errors: primary.errors };
  // Step 2: eligible sources.
  const eligible = new Set(primary.sources.map((s) => s.sourceId));
  const foreign = selectedSourceIds.find((id) => !eligible.has(id));
  if (foreign !== undefined) return { errors: [{ key: "plan.sourceNotAllowed", params: { sourceId: foreign } }] };
  // Step 3: plate-only.
  const n = narrow(primary, selectedSourceIds);
  if (n.keep.length === 0) return { errors: [{ key: "plan.noPlateOnlySource" }] };
  const parts: PlanPart[] = [{
    partId: 0, parentPartId: null, origin: "primary", queryType,
    typeValues: typeValuesOf(primary), values: { ...primary.values },
    sourceIds: n.keep, droppedSourceIds: n.dropped, status: "planned",
  }];
  // Steps 4 and 5: nested parts, in config order; ids are alsoRun index + 1.
  const qt = findQueryType(config, queryType);
  (qt?.alsoRun ?? []).forEach((nested, index) => {
    if (nested.when !== undefined && !holds(config, queryType, nested.when, primary, options)) return;
    const input: Record<string, CanonicalValue> = {};
    for (const [target, source] of Object.entries(nested.fieldMap)) {
      const v = primary.values[source];
      if (v !== undefined) input[target] = v;
    }
    const s = evaluateForm(config, nested.queryType, input, options);
    const base = { partId: index + 1, parentPartId: 0 as const, origin: "alsoRun" as const, queryType: nested.queryType,
      typeValues: typeValuesOf(s), fieldMapApplied: { ...nested.fieldMap } };
    if (s.errors.length > 0) {
      parts.push({ ...base, values: {}, sourceIds: [], droppedSourceIds: [], status: "skipped", skipReasons: s.errors.map(auditReason) });
      return;
    }
    const defaults = s.sources.filter((x) => x.selectedByDefault).map((x) => x.sourceId);
    const m = narrow(s, defaults);
    if (m.keep.length === 0) {
      const key = s.mode === "plateOnly" && defaults.length > 0 ? "plan.noPlateOnlySource" : "plan.nestedNoSources";
      parts.push({ ...base, values: {}, sourceIds: [], droppedSourceIds: [], status: "skipped", skipReasons: [{ key }] });
      return;
    }
    parts.push({ ...base, values: { ...s.values }, sourceIds: m.keep, droppedSourceIds: m.dropped, status: "planned" });
  });
  // Spec 5.2 step 2: total dispatched pairs capped.
  const total = parts.reduce((t, p) => t + p.sourceIds.length, 0);
  if (total > MAX_SOURCES_PER_SUBMIT) return { errors: [{ key: "plan.tooManySources", params: { max: MAX_SOURCES_PER_SUBMIT } }] };
  return { mode: primary.mode, droppedSourceIds: n.dropped, parts };
}
```

`holds` compiles the `NestedQuery.when` against the primary query type with `compileCondition` and evaluates it on `primary.values` (spec 4.2: `NestedQuery.when` reads submitted values, hidden fields absent). Write it with `compileQueryType(config, queryType, now).fieldByKey` and `evaluateCondition(compiled, (k) => primary.values[k] ?? null)`.
